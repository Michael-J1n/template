long long digitDP(long long low, long long high, bool lead_zero = true) {
    string low_s = to_string(low), high_s = to_string(high);
    int n = high_s.size(), diff_lh = n - low_s.size();
    vector memo(n, vector<long long>(12, -1));
    auto dfs = [&](this auto&& dfs, int i, int pre, bool limit_low, bool limit_high) -> long long {
        if (i == n) {
            return 1;
        }
        if (!limit_low && !limit_high && memo[i][pre] >= 0) return memo[i][pre];
        int lo = limit_low && i >= diff_lh ? low_s[i - diff_lh] - '0' : 0;
        int hi = limit_high ? high_s[i] - '0' : 9;
        long long res = 0;
        int d = lo;
        if (lead_zero && limit_low && i < diff_lh) {
            res = dfs(i + 1, 11, true, false);
            d = 1;
        }
        for (; d <= hi; ++d) {
            if (abs(d - pre) < 2) continue;
            res += dfs(i + 1, d, limit_low && d == lo, limit_high && d == hi);
        }
        if (!limit_low && !limit_high) memo[i][pre] = res;
        return res;
    };
    return dfs(0, 11, true, true);
}

bool bipartite(vector<vector<int>>& g, vector<int>& color) {
    int n = g.size();
    color.assign(n, -1);
    auto bfs = [&](int s) -> bool {
        queue<int> q; q.emplace(s);
        color[s] = 0;
        while (!q.empty()) {
            int u = q.front(); q.pop();
            for (auto& v : g[u]) {
                if (color[v] == -1) {
                    color[v] = color[u] ^ 1;
                    q.emplace(v);
                } else if (color[v] == color[u]) {    // u-v-u is an odd cycle
                    return 0;
                }
            }
        }
        return 1;
    };
    for (int i = 0; i < n; ++i) {
        if (color[i] != -1) continue;
        if (!bfs(i)) return 0;
    }
    return 1;
}

struct LazyHeap {
    using ll = long long;
    priority_queue<int, vector<int>, Cmp> pq;
    unordered_map<int, int> erased;
    unordered_map<int, int> cnt;
    void prune() {
        while (!pq.empty()) {
            ll x = pq.top();
            auto it = erased.find(x);
            if (it != erased.end() && it->second > 0) {
                if (--it->second == 0) erased.erase(it);
                pq.pop();
            }
            else break;
        }
    }
    void push(int x) {
        pq.emplace(x);
        ++cnt[x];
    }
    bool erase(int x) {
        auto it = cnt.find(x);
        if (it == cnt.end() || it->second == 0) return false;
        if (--it->second == 0) cnt.erase(it);
        ++erased[x];
        return true;
    }
    void pop() {
        prune();
        if (pq.empty()) return;
        ll x = pq.top(); pq.pop();
        auto it = cnt.find(x);
        if (it != cnt.end() && --it->second == 0) cnt.erase(it);
    }
    int top() {
        prune();
        return pq.top();
    }
    bool empty() {
        prune();
        return cnt.empty();
    }
};

struct LazySegTree {
    static constexpr T INF = numeric_limits<T>::max() / 2;
    struct Node {
        T sum, mx, mn;
    };
    int n;
    Node E = {0, -INF, INF};
    vector<Node> tr;
    vector<T> addTag;
    LazySegTree() {}
    LazySegTree(int _n) : n(_n) {
        tr.assign(n * 4 + 5, {0, 0, 0});
        addTag.assign(n * 4 + 5, 0);
    }
    LazySegTree(const vector<T>& a) : n(a.size()) {
        tr.assign(n * 4 + 5, E);
        addTag.assign(n * 4 + 5, 0);
        build(1, 0, n - 1, a);
    }
    Node merge(const Node& a, const Node& b) {
        T cursum = a.sum + b.sum;
        T curmx = max(a.mx, b.mx);
        T curmn = min(a.mn, b.mn);
        return {cursum, curmx, curmn};
    }
    void pushup(int u) {
        tr[u] = merge(tr[u << 1], tr[u << 1 | 1]);
    }
    void apply_add(int u, int l, int r, T v) {
        tr[u].sum += v * (r - l + 1);
        tr[u].mx += v;
        tr[u].mn += v;
        addTag[u] += v;
    }
    void pushdown(int u, int l, int r) {
        int mid = (l + r) >> 1;
        if (addTag[u]) {
            apply_add(u << 1, l, mid, addTag[u]);
            apply_add(u << 1 | 1, mid + 1, r, addTag[u]);
            addTag[u] = 0;
        }
    }
    void build(int u, int l, int r, const vector<T>& a) {
        if (l == r) {
            tr[u] = {a[l], a[l], a[l]};
            return;
        }
        int mid = (l + r) >> 1;
        build(u << 1, l, mid, a);
        build(u << 1 | 1, mid + 1, r, a);
        pushup(u);
    }
    void add(int u, int l, int r, int ql, int qr, T v) {
        if (ql <= l && r <= qr) {
            apply_add(u, l, r, v);
            return;
        }
        pushdown(u, l, r);
        int mid = (l + r) >> 1;
        if (ql <= mid) add(u << 1, l, mid, ql, qr, v);
        if (qr > mid) add(u << 1 | 1, mid + 1, r, ql, qr, v);
        pushup(u);
    }
    void add(int l, int r, T v) {
        l = max(l, 0), r = min(r, n - 1);
        if (l > r) return;
        add(1, 0, n - 1, l, r, v);
    }
    Node query(int u, int l, int r, int ql, int qr) {
        if (ql <= l && r <= qr) return tr[u];
        pushdown(u, l, r);
        int mid = (l + r) >> 1;
        if (qr <= mid) return query(u << 1, l, mid, ql, qr);
        if (ql > mid) return query(u << 1 | 1, mid + 1, r, ql, qr);
        return merge(query(u << 1, l, mid, ql, qr), query(u << 1 | 1, mid + 1, r, ql, qr));
    }
    Node query(int l, int r) {
        l = max(l, 0), r = min(r, n - 1);
        if (l > r) return E;
        return query(1, 0, n - 1, l, r);
    }

struct blockDecomposition2 {
    int n, B, m;
    vector<long long> a, tag;
    vector<vector<long long>> s;
    blockDecomposition2(const vector<long long>& nums) {
        n = nums.size();
        B = sqrt(n) + 1;
        m = (n + B - 1) / B;
        a = nums;
        tag.assign(m, 0);
        s.resize(m);
        for (int i = 0; i < m; ++i) s[i].reserve(B);
        for (int i = 0; i < n; ++i) {
            int id = i / B;
            s[id].emplace_back(a[i]);
        }
        for (int i = 0; i < m; ++i) ranges::sort(s[i]);
    }
    void rebuild(int b) {
        auto& vec = s[b];
        vec.clear();
        for (int i = b * B; i < min(n, (b + 1) * B); ++i) vec.emplace_back(a[i]);
        ranges::sort(vec);
    }
    void apply(int i, long long v) {
        int b = i / B
        a[i] += v;   
    }
    void add(int l, int r, long long v) {
        int bl = l / B, br = r / B;
        if (bl == br) {
            for (int i = l; i <= r; ++i) apply(i, v);
            rebuild(bl);
        } else {
            for (int i = l; i < (bl + 1) * B; ++i) apply(i, v);
            rebuild(bl);
            for (int b = bl + 1; b < br; ++b) tag[b] += v;
            for (int i = br * B; i <= r; ++i) apply(i, v);
            rebuild(br);
        }
    }
    bool check(int i, long long x, long long y) {
        int b = i / B;
        long long cur = a[i] + tag[b];
        return x <= cur && cur <= y;
    }
    int query(int l, int r, long long x, long long y) {
        int bl = l / B, br = r / B;
        int res = 0;
        if (bl == br) {
            for (int i = l; i <= r; ++i) res += check(i, x, y);
        } else {
            for (int i = l; i < (bl + 1) * B; ++i) res += check(i, x, y);
            for (int b = bl + 1; b < br; ++b) {
                auto& vec = s[b];
                int L = ranges::lower_bound(vec, x - tag[b]) - vec.begin();
                int R = ranges::upper_bound(vec, y - tag[b]) - vec.begin() - 1;
                if (L <= R) res += R - L + 1;
            }
            for (int i = br * B; i <= r; ++i) res += check(i, x, y);
        }
        return res;
    }

        for (int b = B; ~b; --b) {
            int c = int((U(x) >> b) & 1);
            int same = t[u].ch[c], sc = (same == -1 ? 0 : t[same].cnt);
            if (k <= sc) {
                u = same;
            } else {
                k -= sc;
                u = t[u].ch[c ^ 1];
                res |= T(U(1) << b);
            }
        }
        return res;
    }
    T kthXorVal(T x, int k) {
        if (k <= 0 || k > t[0].cnt) return T(-1);
        int u = 0;
        T y = 0;
        for (int b = B; ~b; --b) {
            int c = int((U(x) >> b) & 1);
            int same = t[u].ch[c], sc = (same == -1 ? 0 : t[same].cnt);
            if (k <= sc) {
                y |= T(U(c) << b);
                u = same;
            } else {
                k -= sc;
                y |= T(U(c ^ 1) << b);
                u = t[u].ch[c ^ 1];
            }
        }
        return y;
    }
};

struct Query {
    int l, r, id;
};
vector<int> moAlgorithm(vector<int>& a, vector<Query>& qs) {
    int n = a.size(), q = qs.size();
    vector<int> xs = a;
    sort(xs.begin(), xs.end()); xs.erase(unique(xs.begin(), xs.end()), xs.end());
    int tot = xs.size();
    auto find = [&](int x) -> int { return ranges::lower_bound(xs, x) - xs.begin(); };
    for (int i = 0; i < n; ++i) a[i] = find(a[i]);
    int B = sqrt(n) + 1;  // int B = max(1, int(n / sqrt(q)));
    ranges::sort(qs, [&](const Query& x, const Query& y) {
        int bx = x.l / B, by = y.l / B;
        if (bx != by) return bx < by;
        if (bx & 1) return x.r > y.r;
        return x.r < y.r;
    });
    vector<int> cnt(tot), ans(q);
    int cur = 0;
    auto add = [&](int i) -> void {
        int x = a[i];
        if (cnt[x] == 0) ++cur;
        ++cnt[x];
    };
    auto del = [&](int i) -> void {
        int x = a[i];
        --cnt[x];
        if (cnt[x] == 0) --cur;
    };
    int L = 0, R = -1;
    for (auto& [l, r, id] : qs) {
        while (L > l) add(--L);
        while (R < r) add(++R);
        while (L < l) del(L++);
        while (R > r) del(R--);
        ans[id] = cur;
    }
    return ans;
}

using ll = long long;
using ld = long double;
struct Point {
    ll x, y;
    Point() : x(0), y(0) {}
    Point(ll _x, ll _y) : x(_x), y(_y) {}
    Point operator+(const Point &o) const { return Point(x + o.x, y + o.y); }
    Point operator-(const Point &o) const { return Point(x - o.x, y - o.y); }
    bool operator<(const Point &o) const {
        if (x != o.x) return x < o.x;
        return y < o.y;
    }
    bool operator==(const Point &o) const { return x == o.x && y == o.y; }
};
ld dis(const Point& a, const Point& b) {
    ll dx = a.x - b.x, dy = a.y - b.y;
    return sqrt(dx * dx + dy * dy);
}
ll cross(const Point &a, const Point &b) { return a.x * b.y - a.y * b.x; }
ll cross(const Point &a, const Point &b, const Point &c) { return cross(b - a, c - a); }
ld triangleArea(const Point& a, const Point& b, const Point& c) { return fabsl((ld)cross(a, b, c)) / 2.0; }
vector<Point> convexHull(vector<Point>& p) {
    sort(p.begin(), p.end()); p.erase(unique(p.begin(), p.end()), p.end());
    int n = p.size();
    if (n <= 1) return p;
    vector<Point> s; s.resize(n * 2);
    int top = -1;
    for (int i = 0; i < n; ++i) {
        while (top > 0 && cross(s[top - 1], s[top], p[i]) <= 0) --top;
        s[++top] = p[i];
    }
    int t = top;
    for (int i = n - 2; ~i; --i) {
        while (top > t && cross(s[top - 1], s[top], p[i]) <= 0) --top;
        s[++top] = p[i];
    }
    --top;
    s.resize(top + 1); return s;
}
